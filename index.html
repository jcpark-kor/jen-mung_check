데이터 백업 및 관리

구글 스프레드시트 연동 방법

  1. 새 구글 스프레드시트를 생성합니다.
  2. 상단 메뉴에서 [확장 프로그램] > [Apps Script]를 클릭합니다.
  3. 아래 소스코드를 전체 복사하여 스크립트 에디터에 붙여넣고 저장(Ctrl+S)합니다.
  4. 우측 상단 [배포] > [새 배포]를 누릅니다.
  5. 유형 선택(톱니바퀴) > 웹 앱(Web App)을 선택합니다.
  6. 액세스 권한이 있는 사용자(Who has access)를 [모든 사용자 (Anyone)]로 설정 후 배포합니다.
  7. 생성된 웹 앱 URL을 본 웹사이트의 [설정(⚙️)] 메뉴에 입력하면 완벽히 연동됩니다!
Google Apps Script Code (Code.gs) 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 시간대 이름을 시트명으로 사용 (예: "00:00", "04:00")
    var sheetName = data.timeSlot || "기본시간대";
    var sheet = ss.getSheetByName(sheetName);
    
    // 시트가 존재하지 않는 경우 자동 생성 및 헤더 설정
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["기록시각", "보스명", "상태", "보스유형", "시간대", "메모"]);
      sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#1e293b").setFontColor("#ffffff");
    }
    
    // 데이터 추가 (행 삽입)
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString(),
      data.bossName,
      data.status,
      data.bossType === 'fixed' ? '고정' : '확률',
      data.timeSlot,
      data.memo || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
